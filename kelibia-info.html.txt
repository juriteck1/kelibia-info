function dNext(){
  if(dStep>=5){submitDepot();return;}
  document.getElementById("ds"+dStep).style.display="none";
  document.getElementById("dst"+dStep).className="dstep done";
  dStep++;
  document.getElementById("ds"+dStep).style.display="block";
  document.getElementById("dst"+dStep).className="dstep act";
  document.getElementById("dp-back").style.visibility="visible";
  if(dStep===5)document.getElementById("dp-next").textContent="Soumettre";
}
function dPrev(){
  if(dStep<=1)return;
  document.getElementById("ds"+dStep).style.display="none";
  document.getElementById("dst"+dStep).className="dstep";
  dStep--;
  document.getElementById("ds"+dStep).style.display="block";
  document.getElementById("dst"+dStep).className="dstep act";
  if(dStep===1)document.getElementById("dp-back").style.visibility="hidden";
  document.getElementById("dp-next").textContent="Continuer";
}
